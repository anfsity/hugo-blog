{{- $content := .Content -}} {{- $filePath := .File.Path -}} {{- $dir := path.Dir $filePath -}} {{- $dir = replace $dir "\\" "/" -}} {{- $relDir := strings.TrimPrefix "content/" $dir -}} {{- $imagePrefix := printf "/%s/images/" $relDir -}} {{- $obsidianImgTag := printf `` $imagePrefix -}} {{- $content = $content | replaceRE `!\[\[([^|\]]+)(?:\|[^\]]+)?\]\]` $obsidianImgTag -}} {{- $wrappedTable := printf "
${1}
" -}} {{- $content = $content | replaceRE "((?:.|\n)+?
)" $wrappedTable -}} {{- $content | safeHTML -}}
